\ ANS Forth type system profile (ANSI X3.215-1994 / DPANS94).
\
\ Sources:
\   ANSI X3.215-1994, section 3.1 and table 3.1.
\   Public final-draft HTML: https://www.complang.tuwien.ac.at/forth/dpans-html/dpans3.htm#3.1
\   File types:     https://www.complang.tuwien.ac.at/forth/dpans-html/dpans11.htm#11.3.1
\   Floating types: https://www.complang.tuwien.ac.at/forth/dpans-html/dpans12.htm#12.3.1
\   Search types:   https://www.complang.tuwien.ac.at/forth/dpans-html/dpans16.htm#16.3.1
\
\ `n|u` and `d|ud` are explicit join nodes for the Standard's contextual
\ notations of the same names.  The evaluator has one specification per word,
\ so these nodes provide a conservative representation when signedness is not
\ fixed.  Double-cell and floating values are each represented by one abstract
\ item even when their physical representation occupies another stack or more
\ than one cell.
\
\ Format:
\   type <canonical-name> [alias...]
\   rel <subtype> < <supertype>
\   scanner <name> "<delimiter>"

\ 3.1.3 Single-cell types
type x X cell
type +n nonnegative
type n signed
type u unsigned
type n|u number nu
type char c character
type flag boolean
type true
type false
type xt execution-token
type addr address
type c-addr ca character-address
type a-addr aa aligned-address

\ 3.1.4 Cell-pair types
type xd cell-pair
type d double-signed
type +d double-nonnegative
type ud double-unsigned
type d|ud double-number du

\ 3.1.5 System types
type colon-sys
type do-sys
type case-sys
type of-sys
type orig
type dest
type loop-sys
type nest-sys

\ 11.3.1 File word-set types
type ior io-result
type fam file-access-method
type fileid file-id

\ 12.3.1 Floating-point word-set types
type r float
type f-addr fa float-address
type sf-addr sfa single-float-address
type df-addr dfa double-float-address

\ 16.3.1 Search-order word-set type
type wid word-list-id

\ 3.1.1 standard subtype relationships, with the two join nodes above.
rel +n < n
rel +n < u
rel n < n|u
rel u < n|u
rel n|u < x
rel char < +n
rel true < flag
rel false < flag
rel flag < x
rel xt < x
rel a-addr < c-addr
rel c-addr < addr
rel addr < u
rel +d < d
rel +d < ud
rel d < d|ud
rel ud < d|ud
rel d|ud < xd

\ Word-set types whose values are cells.  I/O results are single-cell numbers.
rel ior < n
rel fam < x
rel fileid < x
rel wid < x

\ Section 12.3.1 defines all three as subsets of aligned addresses.
rel f-addr < a-addr
rel sf-addr < a-addr
rel df-addr < a-addr

scanner EOL "\n"
